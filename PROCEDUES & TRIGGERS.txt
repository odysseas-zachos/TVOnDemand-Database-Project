DROPPROCEDUREIFEXISTS last_names;

 DELIMITER $

CREATEPROCEDURE last_names(IN lastname_a VARCHAR(255), IN lastname_b VARCHAR(255))
BEGIN
SELECT first_name, last_name FROM actor
WHERE  last_name  >= CONCAT(lastname_a, '%')  AND last_name <CONCAT(lastname_b, '%')
ORDER BY last_name ASC;
END $

 DELIMITER ;

DROPPROCEDUREIFEXISTS same_last_names;

DELIMITER $

CREATEPROCEDURE same_last_names (IN surname varchar(45))

BEGIN
DECLARE lastname varchar(45);
DECLARE done BOOLEANDEFAULT false;
DECLAREtimesINT;

DECLARE curLastName
CURSOR FOR
SELECTcount(actor_id),last_name from actor WHERE last_name = surname GROUP BY last_name;


DECLARECONTINUE HANDLER
        FOR NOT FOUND SET done = true;

OPEN curLastName;

REPEAT
IF (NOT done) THEN
FETCH curLastName INTOtimes,lastname;
ENDIF;
    UNTIL done ENDREPEAT;

CLOSE curLastName;

SELECTCONCAT("Last name ",lastname," found ",times," times") AS Result;

END $

DELIMITER ;

IF (old.active <> new.active) THEN
        SIGNAL SQLSTATE VALUE'45000'SET MESSAGE_TEXT = 'Status can not be changed to zero since user is active!';
ENDIF;
ENDIF;
END $
delimiter ;

delimiter $

CREATE TRIGGER trigger4_2 AFTERUPDATEON movie_payment
for each row
begin
DECLARE discount DECIMAL(5,2);
DECLARE discount_date datetime;
DECLARE rental_count int;
DECLARE rental_amount decimal(5,2);
IF (SELECT rental_id FROM MOVIE_RENTAL WHERESUBSTRING(rental_date,1,10) LIKESUBSTRING(current_timestamp,1,10)) THEN
SET discount_date = SUBSTRING(current_timestamp,1,10);
ENDIF;
SET rental_count = (SELECTCOUNT(customer_id) FROM MOVIE_RENTAL WHERE old.customer_id = new.customer_id GROUP BY customer_id);
IF (rental_count >2ANDSUBSTRING(rental_date,1,10) LIKE discount_date) THEN
SET rental_amount = (SELECT amount FROM movie_payment WHERE old.customer_id = new.customer_id);
SET discount = rental_amount/2;
INSERTINTO movie_payment (payment_id, customer_id, rental_id, amount, payment_date) VALUES (new.payment_id, new.customer_id, new.rental_id, discount, current_timestamp);
ENDIF;
END $
